    Current code only handles contact deep links. Add a clause for queue:

    ```javascript
    function checkDeepLinkQueryParams() {
      const params = new URLSearchParams(window.location.search);
      const view = params.get('view');
      const id = params.get('id');
      if (view === 'contact' && id) {
        history.replaceState(null, '', window.location.pathname);
        window.location.hash = `/contact/${id}`;
        return true;
      }
      if (view === 'queue') {
        history.replaceState(null, '', window.location.pathname);
        window.location.hash = '/queue';
        return true;
      }
      return false;
    }
    ```

    This ensures the "Review in App" CTA from the email (`?view=queue`) correctly routes to the queue page without leaving query params in the URL bar.
